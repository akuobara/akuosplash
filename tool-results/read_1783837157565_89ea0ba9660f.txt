   318→   318→  // Helper: aplicar precio según método de pago
   319→   319→  function aplicarPrecioSegunMetodo(metodo: string, cliente: ClienteOption | null) {
   320→   320→    const precioEfectivo = cliente?.precioEfectivoPorLitro || config.precioEfectivoPorLitro || 0;
   321→   321→    const precioTransferencia = cliente?.precioTransferenciaPorLitro || config.precioTransferenciaPorLitro || 0;
   322→   322→    let precio = 0;
   323→   323→    if (metodo === "efectivo") {
   324→   324→      precio = precioEfectivo;
   325→   325→    } else if (metodo === "transferencia") {
   326→   326→      precio = precioTransferencia;
   327→   327→    } else {
   328→   328→      // mixto o crédito (sin pago): usar el precio MÁS ALTO como base
   329→   329→      // — para "sin pago" esto genera la deuda correcta desde el inicio,
   330→   330→      //   y el usuario puede ajustar el precio al momento de cobrar
   331→   331→      precio = Math.max(precioEfectivo, precioTransferencia);
   332→   332→    }
   333→   333→    // Siempre setear el precio (incluso si es 0, para que el usuario vea que no hay configurado)
   334→   334→    setPrecioPorLitro(precio > 0 ? String(precio) : "");
   335→   335→  }

   345→   345→      metodoAnterior === "efectivo"
   346→   346→        ? clienteActual?.precioEfectivoPorLitro || config.precioEfectivoPorLitro
   347→   347→        : metodoAnterior === "transferencia"
   348→   348→        ? clienteActual?.precioTransferenciaPorLitro || config.precioTransferenciaPorLitro
   349→   349→        : clienteActual?.precioEfectivoPorLitro || config.precioEfectivoPorLitro;
   350→   350→    const precioActual = Number(precioPorLitro) || 0;
   351→   351→    if (precioActual === precioEsperadoAnterior || precioActual === 0) {
   352→   352→      aplicarPrecioSegunMetodo(nuevoMetodo, clienteActual);
   353→   353→    }
   354→   354→  }

   358→   358→    lecturaAnterior !== "" && lecturaActual !== ""
   359→   359→      ? Number(lecturaActual) - Number(lecturaAnterior)
   360→   360→      : null;
   361→   361→
   362→   362→  const litrosFinal = litrosManual !== "" ? Number(litrosManual) : litrosCalculado ?? 0;
   363→   363→  const precioFinal = Number(precioPorLitro) || 0;
   364→   364→  const totalCalculado = litrosFinal * precioFinal;
   365→   365→  const efectivoFinal = Number(montoEfectivo) || 0;
   366→   366→  const transferenciaFinal = Number(montoTransferencia) || 0;
   367→   367→  const recibidoFinal = efectivoFinal + transferenciaFinal;
   368→   368→  const saldoFinal = totalCalculado - recibidoFinal;
   369→   369→
   370→   370→  // Auto-actualizar estadoPago según monto recibido (SIEMPRE automático, no manual)
   371→   371→  useEffect(() => {
   372→   372→    if (recibidoFinal === 0) {

   374→   374→    } else if (recibidoFinal >= totalCalculado && totalCalculado > 0) {

   376→   376→    } else {
   377→   377→      setEstadoPago("parcial");
   378→   378→    }
   379→   379→  }, [recibidoFinal, totalCalculado]);
