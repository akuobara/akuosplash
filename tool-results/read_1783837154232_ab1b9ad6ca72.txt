   318→  // Helper: aplicar precio según método de pago
   319→  function aplicarPrecioSegunMetodo(metodo: string, cliente: ClienteOption | null) {
   320→    const precioEfectivo = cliente?.precioEfectivoPorLitro || config.precioEfectivoPorLitro || 0;
   321→    const precioTransferencia = cliente?.precioTransferenciaPorLitro || config.precioTransferenciaPorLitro || 0;
   322→    let precio = 0;
   323→    if (metodo === "efectivo") {
   324→      precio = precioEfectivo;
   325→    } else if (metodo === "transferencia") {
   326→      precio = precioTransferencia;
   327→    } else {
   328→      // mixto o crédito (sin pago): usar el precio MÁS ALTO como base
   329→      // — para "sin pago" esto genera la deuda correcta desde el inicio,
   330→      //   y el usuario puede ajustar el precio al momento de cobrar
   331→      precio = Math.max(precioEfectivo, precioTransferencia);
   332→    }
   333→    // Siempre setear el precio (incluso si es 0, para que el usuario vea que no hay configurado)
   334→    setPrecioPorLitro(precio > 0 ? String(precio) : "");
   335→  }

   337→  // Si cambia método de pago (y el usuario no ha tocado el precio manualmente)
   338→  function onChangeMetodoPago(nuevoMetodo: string) {
   339→    const metodoAnterior = metodoPago;
   340→    setMetodoPago(nuevoMetodo);

   343→    const clienteActual = clienteId ? clientes.find((c) => c.id === clienteId) : null;
   344→    const precioEsperadoAnterior =
   345→      metodoAnterior === "efectivo"
   346→        ? clienteActual?.precioEfectivoPorLitro || config.precioEfectivoPorLitro
   347→        : metodoAnterior === "transferencia"
   348→        ? clienteActual?.precioTransferenciaPorLitro || config.precioTransferenciaPorLitro
   349→        : clienteActual?.precioEfectivoPorLitro || config.precioEfectivoPorLitro;
   350→    const precioActual = Number(precioPorLitro) || 0;
   351→    if (precioActual === precioEsperadoAnterior || precioActual === 0) {
   352→      aplicarPrecioSegunMetodo(nuevoMetodo, clienteActual);
   353→    }
   354→  }

   358→    lecturaAnterior !== "" && lecturaActual !== ""

   360→      : null;
   361→
   362→  const litrosFinal = litrosManual !== "" ? Number(litrosManual) : litrosCalculado ?? 0;

   364→  const totalCalculado = litrosFinal * precioFinal;
   365→  const efectivoFinal = Number(montoEfectivo) || 0;
   366→  const transferenciaFinal = Number(montoTransferencia) || 0;
   367→  const recibidoFinal = efectivoFinal + transferenciaFinal;
   368→  const saldoFinal = totalCalculado - recibidoFinal;
   369→
   370→  // Auto-actualizar estadoPago según monto recibido (SIEMPRE automático, no manual)
   371→  useEffect(() => {
   372→    if (recibidoFinal === 0) {
   373→      setEstadoPago("pendiente");
   374→    } else if (recibidoFinal >= totalCalculado && totalCalculado > 0) {
   375→      setEstadoPago("pagado");
   376→    } else {
   377→      setEstadoPago("parcial");
   378→    }
   379→  }, [recibidoFinal, totalCalculado]);

   394→  // Guarda la entrega con un vueltaId opcional (cuando se inicia/reabre desde el diálogo)
   395→  async function guardarEntrega(vueltaIdOverride?: string) {
   396→    // Capturar el estado actual en variables locales para evitar problemas de stale closure
   397→    const modoEst = modoEstanques;
   398→    const estanques = estanquesEntrega;
   399→    const clienteNombreLocal = clienteNombre.trim();
   400→    const litrosFinalLocal = litrosFinal;
   401→    const precioFinalLocal = precioFinal;
   402→    const totalCalculadoLocal = totalCalculado;
   403→    const efectivoFinalLocal = efectivoFinal;
   404→    const transferenciaFinalLocal = transferenciaFinal;
   405→
   406→    // Debug: trace what we're about to save
   407→    console.log("[guardarEntrega] estado capturado:", {
   408→      modoEstanques: modoEst,
   409→      estanquesCount: estanques.length,
   410→      estanquesConLitros: estanques.filter(e => Number(e.litros) > 0).length,
   411→      clienteNombre: clienteNombreLocal,
   412→      litrosFinal: litrosFinalLocal,
   413→      vueltaIdOverride,
   414→    });
   415→
   416→    // Determinar si hay datos de estanques con litros > 0
   417→    const estanquesConLitros = estanques.filter((e) => Number(e.litros) > 0);
   418→
   419→    // Si está en modo estanques (o hay estanques con litros), usar el path de estanques
   420→    if ((modoEst || estanquesConLitros.length > 0) && estanques.length > 0) {
   421→      const totalLitrosEst = estanquesConLitros.reduce((s, e) => s + (Number(e.litros) || 0), 0);
   422→      if (totalLitrosEst <= 0) {
   423→        toast.error("Ingresa al menos litros en un estanque");
   424→        return;
   425→      }
   426→      const totalVentaEst = estanquesConLitros.reduce((s, e) => s + (Number(e.litros) || 0) * (Number(e.precio) || 0), 0);
   427→      const totalEfectivoEst = estanquesConLitros.filter((e) => e.metodoPago === "efectivo").reduce((s, e) => s + (Number(e.montoRecibido) || 0), 0);
   428→      const totalTransferenciaEst = estanquesConLitros.filter((e) => e.metodoPago === "transferencia").reduce((s, e) => s + (Number(e.montoRecibido) || 0), 0);
   429→      const totalRecibidoEst = totalEfectivoEst + totalTransferenciaEst;
   430→
   431→      let estadoPagoEst = "pendiente";
   432→      if (totalRecibidoEst >= totalVentaEst && totalVentaEst > 0) estadoPagoEst = "pagado";
   433→      else if (totalRecibidoEst > 0) estadoPagoEst = "parcial";
   434→
   435→      setSaving(true);
   436→      try {
   437→        const payload: Record<string, unknown> = {
   438→          clienteId: clienteId && clienteId !== "__manual" ? clienteId : null,
   439→          clienteNombre: clienteNombreLocal,
   440→          clienteDireccion: clienteDireccion.trim() || null,
   441→          clienteTelefono: clienteTelefono.trim() || null,
   442→          clienteZona: clienteZona.trim() || null,
   443→          litros: totalLitrosEst,
   444→          metodoPago: "mixto",
   445→          precioPorLitro: 0,
   446→          total: totalVentaEst,
   447→          montoEfectivo: totalEfectivoEst,
   448→          montoTransferencia: totalTransferenciaEst,
   449→          estadoPago: estadoPagoEst,
   450→          fotoMedidorUrl: fotoUrl,
   451→          nota: nota.trim() || null,
   452→          detalleEstanquesJson: JSON.stringify(estanquesConLitros.map((e) => ({
   453→            nombre: e.nombre,
   454→            litros: Number(e.litros) || 0,
   455→            precioPorLitro: Number(e.precio) || 0,
   456→            total: (Number(e.litros) || 0) * (Number(e.precio) || 0),
   457→            metodoPago: e.metodoPago,
   458→            montoRecibido: Number(e.montoRecibido) || 0,
   459→          }))),
   460→        };
   461→        if (vueltaIdOverride) payload.vueltaId = vueltaIdOverride;
   462→        if (pedidoSeleccionadoId) payload.pedidoId = pedidoSeleccionadoId;
   463→
   464→        console.log("[guardarEntrega] guardando con estanques, payload:", {
