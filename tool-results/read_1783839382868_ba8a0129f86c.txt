   440→
   441→      setSaving(true);
   442→      try {

   457→          nota: nota.trim() || null,

   459→            nombre: e.nombre,
   460→            litros: Number(e.litros) || 0,
   461→            precioPorLitro: Number(e.precio) || 0,
   462→            total: (Number(e.litros) || 0) * (Number(e.precio) || 0),
   463→            metodoPago: e.metodoPago,
   464→            montoRecibido: Number(e.montoRecibido) || 0,

   468→        if (pedidoSeleccionadoId) payload.pedidoId = pedidoSeleccionadoId;
   469→
   470→        if (isEdit && entrega) {
   471→          await apiPatch(`/api/entregas/${entrega.id}`, payload);

   473→        } else {
   474→          await apiPost("/api/entregas", payload);
   475→          toast.success("Entrega registrada");
   476→        }
   477→        onSaved();
   478→        onOpenChange(false);
   479→        setDialogoVueltaAbierto(false);
   480→      } catch (e) {
   481→        console.error("[guardarEntrega] error al guardar:", e);
   482→        toast.error(e instanceof Error ? e.message : "No se pudo guardar");
   483→      } finally {
   484→        setSaving(false);
   485→      }
   486→      return;
   487→    }
   488→
   489→    if (litrosFinalLocal <= 0) {
   490→      toast.error("Los litros deben ser mayores a 0");
   491→      return;
   492→    }
   493→
   494→    setSaving(true);
   495→    try {
   496→      const payload: Record<string, unknown> = {
   497→        clienteId: clienteId && clienteId !== "__manual" ? clienteId : null,
   498→        clienteNombre: clienteNombreLocal,
   499→        clienteDireccion: clienteDireccion.trim() || null,
   500→        clienteTelefono: clienteTelefono.trim() || null,
   501→        clienteZona: clienteZona.trim() || null,
   502→        lecturaAnterior: lecturaAnterior !== "" ? Number(lecturaAnterior) : null,
   503→        lecturaActual: lecturaActual !== "" ? Number(lecturaActual) : null,
   504→        litros: litrosFinalLocal,
   505→        metodoPago,
   506→        precioPorLitro: precioFinalLocal,
   507→        total: totalCalculadoLocal,
   508→        montoEfectivo: efectivoFinalLocal,
   509→        montoTransferencia: transferenciaFinalLocal,
   510→        estadoPago,
   511→        fotoMedidorUrl: fotoUrl,
   512→        nota: nota.trim() || null,
   513→      };
   514→      if (vueltaIdOverride) payload.vueltaId = vueltaIdOverride;
   515→      if (pedidoSeleccionadoId) payload.pedidoId = pedidoSeleccionadoId;
   516→
   517→      if (isEdit && entrega) {
   518→        await apiPatch(`/api/entregas/${entrega.id}`, payload);
   519→        toast.success("Entrega actualizada");
   520→      } else {
   521→        await apiPost("/api/entregas", payload);
   522→        toast.success("Entrega registrada");
   523→      }
   524→      onSaved();
   525→      onOpenChange(false);
   526→      setDialogoVueltaAbierto(false);

   528→      console.error("[guardarEntrega] error al guardar:", e);

   619→  async function handleSubmit(e: React.FormEvent) {
   620→    e.preventDefault();
   621→    if (!clienteNombre.trim()) {
   622→      toast.error("Indica el nombre del cliente");
   623→      return;
   624→    }
   625→
   626→    // Si no hay vuelta activa, validar litros ANTES de abrir el diálogo
   627→    if (!vueltaActual) {
   628→      // Verificar si hay camiones disponibles
   629→      if (!camiones || camiones.filter(c => c.activo).length === 0) {
   630→        toast.error("No hay vuelta activa ni camiones registrados. Agrega un camión en Ajustes primero.", { duration: 8000 });
   631→        return;
   632→      }
   633→      // Validar litros antes de abrir el diálogo para evitar vueltas huérfanas
   634→      if (!validarLitrosAntesDeVuelta()) return;
   635→      // Cargar vueltas recientes y abrir el diálogo
   636→      await cargarVueltasRecientes();
   637→      setDialogoVueltaAbierto(true);
   638→      return;
   639→    }
   640→
   641→    // Calcular litros totales según modo
   642→    let litrosAEntregar = litrosFinal;
   643→    if (modoEstanques && estanquesEntrega.length > 0) {
   644→      litrosAEntregar = estanquesEntrega.reduce((s, e) => s + (Number(e.litros) || 0), 0);
   645→    }
   646→
   647→    // Avisar (no bloquear) si excede la capacidad del camión
   648→    if (vueltaActual.capacidadCamion && vueltaActual.capacidadCamion > 0) {
   649→      const disponible = vueltaActual.capacidadCamion - vueltaActual.litros;
   650→      const tolerancia = 300;
   651→      if (litrosAEntregar > disponible + tolerancia) {
   652→        if (disponible <= 0) {
   653→          toast.warning(
   654→            `El camión está vacío. Esta entrega se registrará pero deberías cerrar la vuelta #${vueltaActual.numero} y empezar una nueva.`,
   655→            { duration: 8000 }
   656→          );
   657→        } else {
   658→          toast.warning(
   659→            `Quedan ${formatNumber(disponible, 0)} L en el camión. Entregarás ${formatNumber(litrosAEntregar, 0)} L. ` +
   660→            `Cierra la vuelta y empieza una nueva para recargar.`,
   661→            { duration: 8000 }
   662→          );
   663→        }
   664→      } else if (disponible - litrosAEntregar <= tolerancia && disponible - litrosAEntregar > 0) {
   665→        toast.info(
   666→          `Después de esta entrega quedarán ${formatNumber(disponible - litrosAEntregar, 0)} L en el camión. ` +
   667→          `Considera cerrar la vuelta pronto.`,
   668→          { duration: 6000 }
   669→        );
   670→      }
   671→    }
   672→
   673→    // Guardar con la vuelta activa actual
   674→    await guardarEntrega();
   675→  }
