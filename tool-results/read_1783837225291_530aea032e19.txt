   351→  async function handleSave(e: React.FormEvent) {
   352→    e.preventDefault();

   364→      return;
   365→    }
   366→
   367→    setSaving(true);
   368→    try {
   369→      const payload = {
   370→        clienteId: clienteId === "__nuevo" ? null : clienteId,
   371→        clienteNombre: clienteNombre.trim(),
   372→        clienteDireccion: clienteDireccion.trim() || null,
   373→        clienteTelefono: clienteTelefono.trim() || null,
   374→        clienteZona: clienteZona.trim() || null,

   378→        indicacionesAcceso: indicacionesAcceso.trim() || null,

   380→        asignacionTipo,

   388→        await apiPost("/api/pedidos", payload);

   390→      }

   411→  async function handleConvertir(e: React.FormEvent) {
   412→    e.preventDefault();
   413→    if (!convirtiendo) return;
   414→
   415→    // Si está en modo estanques, calcular totales desde el desglose
   416→    if (modoEstanquesConvertir && estanquesConvertir.length > 0) {
   417→      const totalLitrosEst = estanquesConvertir.reduce((s, e) => s + (Number(e.litros) || 0), 0);
   418→      if (totalLitrosEst <= 0) {
   419→        toast.error("Ingresa al menos litros en un estanque");
   420→        return;
   421→      }
   422→      const totalVentaEst = estanquesConvertir.reduce((s, e) => s + (Number(e.litros) || 0) * (Number(e.precio) || 0), 0);
   423→      const totalEfectivoEst = estanquesConvertir.filter((e) => e.metodoPago === "efectivo").reduce((s, e) => s + (Number(e.montoRecibido) || 0), 0);
   424→      const totalTransferenciaEst = estanquesConvertir.filter((e) => e.metodoPago === "transferencia").reduce((s, e) => s + (Number(e.montoRecibido) || 0), 0);
   425→
   426→      setSaving(true);
   427→      try {
   428→        await apiPost(`/api/pedidos/${convirtiendo.id}/convertir`, {
   429→          litrosReales: totalLitrosEst,
   430→          precioPorLitro: 0,
   431→          montoEfectivo: totalEfectivoEst,
   432→          montoTransferencia: totalTransferenciaEst,
   433→          detalleEstanquesJson: JSON.stringify(estanquesConvertir.map((e) => ({
   434→            nombre: e.nombre,
   435→            litros: Number(e.litros) || 0,
   436→            precioPorLitro: Number(e.precio) || 0,
   437→            total: (Number(e.litros) || 0) * (Number(e.precio) || 0),
   438→            metodoPago: e.metodoPago,
   439→            montoRecibido: Number(e.montoRecibido) || 0,
   440→          }))),
   441→        });
   442→        toast.success("Pedido convertido en entrega");
   443→        setConvirtiendo(null);
   444→        onRefresh();
   445→      } catch (e) {
   446→        toast.error(e instanceof Error ? e.message : "No se pudo convertir");
   447→      } finally {
   448→        setSaving(false);
   449→      }
   450→      return;
   451→    }
   452→
   453→    setSaving(true);
   454→    try {
   455→      await apiPost(`/api/pedidos/${convirtiendo.id}/convertir`, {
   456→        litrosReales: Number(litrosReales) || convirtiendo.litrosSolicitados,
   457→        precioPorLitro: Number(precioPorLitro) || undefined,
   458→        montoEfectivo: Number(montoEfectivo) || 0,
   459→        montoTransferencia: Number(montoTransferencia) || 0,
   460→      });
   461→      toast.success("Pedido convertido en entrega");
   462→      setConvirtiendo(null);
   463→      onRefresh();
   464→    } catch (e) {
   465→      toast.error(e instanceof Error ? e.message : "No se pudo convertir");
   466→    } finally {
   467→      setSaving(false);
   468→    }
   469→  }
