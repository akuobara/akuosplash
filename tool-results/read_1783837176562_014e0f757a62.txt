   143→  const [requiereClientePresente, setRequiereClientePresente] = useState(true);
   144→  const [indicacionesAcceso, setIndicacionesAcceso] = useState("");
   145→  const [usoAgua, setUsoAgua] = useState("consumo");
   146→  const [usoAguaNota, setUsoAguaNota] = useState("");
   147→  const [estanques, setEstanques] = useState<{ tipo: string; capacidad: string; nombre: string; ubicacion: string; facturaSeparada: boolean }[]>([]);
   148→  const [choferPreferidoId, setChoferPreferidoId] = useState<string>("");
   149→  const [mapPickerOpen, setMapPickerOpen] = useState(false);

   151→

   173→      c.nombre.toLowerCase().includes(q) ||
   174→      (c.direccion ?? "").toLowerCase().includes(q) ||
   175→      (c.telefono ?? "").toLowerCase().includes(q) ||
   176→      (c.zona ?? "").toLowerCase().includes(q)

   186→    setEmail("");

   189→    setPrecioTransferencia("");

   208→    setRut(c.rut ?? "");
   209→    setDireccion(c.direccion ?? "");

   214→    setPrecioTransferencia(c.precioTransferenciaPorLitro ? String(c.precioTransferenciaPorLitro) : "");

   216→    setLat(c.lat != null ? String(c.lat) : "");
   217→    setLng(c.lng != null ? String(c.lng) : "");
   218→    setRequiereIngresoPropiedad(c.requiereIngresoPropiedad ?? false);
   219→    setEntregaDesdeAfuera(c.entregaDesdeAfuera ?? false);
   220→    setConexionExterna(c.conexionExterna ?? false);
   221→    setRequiereClientePresente(c.requiereClientePresente ?? true);
   222→    setIndicacionesAcceso(c.indicacionesAcceso ?? "");
   223→    setUsoAgua(c.usoAgua || "consumo");
   224→    setUsoAguaNota(c.usoAguaNota ?? "");
   225→    try {
   226→      setEstanques(c.estanquesJson ? JSON.parse(c.estanquesJson) : []);
   227→    } catch { setEstanques([]); }
   228→    setChoferPreferidoId(c.choferPreferidoId ?? "");
   229→    setFormOpen(true);
   230→  }

   232→  // Abrir detalle del cliente con su historial
   233→  async function abrirDetalle(c: Cliente) {
   234→    setClienteDetalle(c);
   235→    setDetalleTab("info");
   236→    setDetalleLoading(true);
   237→    try {
   238→      const [ents, peds, pays] = await Promise.all([
   239→        apiGet<{ entregas: any[] }>(`/api/entregas?period=todo&clienteId=${c.id}`),
   240→        apiGet<{ pedidos: any[] }>(`/api/pedidos?clienteId=${c.id}`),
   241→        apiGet<{ pagos: any[] }>(`/api/pagos?clienteId=${c.id}`),
   242→      ]);
   243→      setDetalleEntregas(ents.entregas || []);
   244→      setDetallePedidos(peds.pedidos || []);
   245→      setDetallePagos(pays.pagos || []);
   246→    } catch (e) {
   247→      console.error(e);
   248→    } finally {
   249→      setDetalleLoading(false);
   250→    }
   251→  }

   253→  // Capturar ubicación GPS actual del dispositivo
   254→  function obtenerMiUbicacion() {
   255→    if (!navigator.geolocation) {
   256→      toast.error("Tu navegador no soporta geolocalización");
   257→      return;
   258→    }
   259→    setObteniendoGPS(true);
   260→    navigator.geolocation.getCurrentPosition(
   261→      (pos) => {
   262→        setLat(pos.coords.latitude.toFixed(6));
   263→        setLng(pos.coords.longitude.toFixed(6));
   264→        setObteniendoGPS(false);
   265→        toast.success("Ubicación capturada");
   266→      },
   267→      (err) => {
   268→        setObteniendoGPS(false);
   269→        toast.error(err.message || "No se pudo obtener tu ubicación");
   270→      },
   271→      { enableHighAccuracy: true, timeout: 10000, maximumAge: 0 }
   272→    );
   273→  }
   274→
   275→  async function handleSave(e: React.FormEvent) {
   276→    e.preventDefault();
   277→    if (!nombre.trim()) {
   278→      toast.error("El nombre es obligatorio");
   279→      return;
   280→    }
   281→    setSaving(true);
   282→    try {
   283→      const payload = {
   284→        nombre: nombre.trim(),
   285→        rut: rut.trim() || null,
   286→        direccion: direccion.trim() || null,
   287→        telefono: telefono.trim() || null,
   288→        email: email.trim() || null,
   289→        zona: zona.trim() || null,
   290→        precioEfectivoPorLitro: Number(precioEfectivo) || 0,
   291→        precioTransferenciaPorLitro: Number(precioTransferencia) || 0,
   292→        nota: nota.trim() || null,
   293→        lat: lat.trim() === "" ? null : Number(lat),
   294→        lng: lng.trim() === "" ? null : Number(lng),
   295→        requiereIngresoPropiedad,
   296→        entregaDesdeAfuera,
   297→        conexionExterna,
   298→        requiereClientePresente,
   299→        indicacionesAcceso: indicacionesAcceso.trim() || null,
   300→        usoAgua,
   301→        usoAguaNota: usoAguaNota.trim() || null,
   302→        estanquesJson: estanques.length > 0 ? JSON.stringify(estanques) : null,
   303→        choferPreferidoId: choferPreferidoId || null,
   304→      };
   305→      if (editando) {
   306→        await apiPatch(`/api/clientes/${editando.id}`, payload);
   307→        toast.success("Cliente actualizado");
   308→      } else {
   309→        await apiPost("/api/clientes", payload);
   310→        toast.success("Cliente creado");
   311→      }
   312→      setFormOpen(false);
   313→      onRefresh();
   314→    } catch (e) {
   315→      toast.error(e instanceof Error ? e.message : "No se pudo guardar");
   316→    } finally {
   317→      setSaving(false);
   318→    }
   319→  }
   320→
   321→  async function handleBorrar(c: Cliente) {
   322→    try {
   323→      await apiDelete(`/api/clientes/${c.id}`);
   324→      toast.success("Cliente eliminado");
   325→      setBorrando(null);
   326→      onRefresh();
   327→    } catch (e) {
   328→      toast.error(e instanceof Error ? e.message : "No se pudo eliminar");
   329→    }
   330→  }
   331→
   332→  async function handlePago(e: React.FormEvent) {
   333→    e.preventDefault();
   334→    if (!clienteDetalle) return;
   335→    const monto = Number(pagoMonto) || 0;
   336→    if (monto <= 0) {
   337→      toast.error("El monto debe ser mayor a 0");
   338→      return;
   339→    }
   340→    setPagoSaving(true);
   341→    try {
   342→      await apiPost("/api/pagos", {
   343→        clienteId: clienteDetalle.id,
   344→        monto,
   345→        metodoPago: pagoMetodo,
   346→        nota: pagoNota.trim() || null,
   347→      });
   348→      toast.success("Pago registrado");
   349→      setPagoOpen(false);
   350→      setPagoMonto("");
   351→      setPagoNota("");
   352→      // Recargar detalle
   353→      abrirDetalle(clienteDetalle);
   354→      onRefresh();
   355→    } catch (e) {
   356→      toast.error(e instanceof Error ? e.message : "No se pudo registrar el pago");
   357→    } finally {
   358→      setPagoSaving(false);
   359→    }
   360→  }

   362→  return (
   363→    <div className="space-y-4">
   364→      {/* Encabezado */}

   402→                : "No se encontraron clientes."}

   415→                  <div className="min-w-0 flex-1">
   416→                    <p className="truncate font-semibold">{c.nombre}</p>
   417→                    {c._count && (
   418→                      <p className="text-xs text-muted-foreground">

   421→                    )}

   423→                  {!readOnly && (
   424→                    <DropdownMenu>
   425→                      <DropdownMenuTrigger asChild>
   426→                        <Button variant="ghost" size="icon" className="h-7 w-7 shrink-0" onClick={(e) => e.stopPropagation()}>
   427→                          <MoreVertical className="h-4 w-4" />
   428→                        </Button>
   429→                      </DropdownMenuTrigger>
   430→                      <DropdownMenuContent align="end" onClick={(e) => e.stopPropagation()}>

   442→                        </DropdownMenuItem>
   443→                      </DropdownMenuContent>
   444→                    </DropdownMenu>
   445→                  )}
   446→                </div>
   447→                <div className="space-y-1 text-sm text-muted-foreground">
   448→                  {c.zona && (
   449→                    <div className="flex items-center gap-1.5">
   450→                      <MapPin className="h-3.5 w-3.5 shrink-0" />
   451→                      <Badge variant="outline" className="text-[10px]">{c.zona}</Badge>
   452→                    </div>
   453→                  )}
   454→                  {c.direccion && (
   455→                    <div className="flex items-start gap-1.5">
   456→                      <MapPin className="mt-0.5 h-3.5 w-3.5 shrink-0" />
   457→                      <span className="truncate">{c.direccion}</span>
   458→                    </div>
   459→                  )}
   460→                  {c.telefono && (
   461→                    <div className="flex items-center gap-1.5">
   462→                      <Phone className="h-3.5 w-3.5 shrink-0" />
   463→                      <a href={`tel:${c.telefono}`} className="hover:underline">
   464→                        {c.telefono}
   465→                      </a>
   466→                    </div>
   467→                  )}
   468→                </div>
   469→                {/* Precios por método */}
   470→                {(c.precioEfectivoPorLitro > 0 || c.precioTransferenciaPorLitro > 0) && (
   471→                  <div className="mt-1 flex flex-wrap gap-2">
   472→                    {c.precioEfectivoPorLitro > 0 && (
   473→                      <Badge variant="secondary" className="text-[11px]">
   474→                        <Banknote className="mr-1 h-3 w-3" />
   475→                        {formatCurrency(c.precioEfectivoPorLitro)}/L
   476→                      </Badge>
   477→                    )}
   478→                    {c.precioTransferenciaPorLitro > 0 && (
   479→                      <Badge variant="secondary" className="text-[11px]">
   480→                        <Smartphone className="mr-1 h-3 w-3" />
   481→                        {formatCurrency(c.precioTransferenciaPorLitro)}/L
   482→                      </Badge>
   483→                    )}
   484→                  </div>
   485→                )}
   486→
   487→                {/* Indicadores de acceso */}
   488→                {(c.requiereIngresoPropiedad || c.entregaDesdeAfuera || c.conexionExterna || c.requiereClientePresente === false) && (
   489→                  <div className="mt-1 flex flex-wrap gap-1">
   490→                    {c.requiereIngresoPropiedad && (
   491→                      <Badge variant="outline" className="text-[10px] gap-0.5" title="Hay que ingresar a la propiedad">
   492→                        <DoorOpen className="h-2.5 w-2.5" /> Ingreso
   493→                      </Badge>
   494→                    )}
   495→                    {c.entregaDesdeAfuera && (
   496→                      <Badge variant="outline" className="text-[10px] gap-0.5" title="Se puede entregar desde afuera">
   497→                        <Fence className="h-2.5 w-2.5" /> Desde afuera
   498→                      </Badge>
   499→                    )}
   500→                    {c.conexionExterna && (
   501→                      <Badge variant="outline" className="text-[10px] gap-0.5 text-emerald-700 dark:text-emerald-300" title="Conexión/llave externa">
   502→                        <KeyRound className="h-2.5 w-2.5" /> Conexión
   503→                      </Badge>
