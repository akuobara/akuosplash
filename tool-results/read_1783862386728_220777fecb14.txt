   417→        onOpenChange={setConfigOpen}
   418→        config={config}
   419→        onSaved={onRefresh}
   420→      />
   421→    </div>

   423→}

   486→                  <p className="mb-1 text-xs uppercase text-muted-foreground">Impuestos</p>

   671→function ConfigModal({
   672→  open,
   673→  onOpenChange,
   674→  config,
   675→  onSaved,
   676→}: {
   677→  open: boolean;
   678→  onOpenChange: (o: boolean) => void;
   679→  config: Config | null;
   680→  onSaved: () => void;
   681→}) {
   682→  const [modo, setModo] = useState("porcentaje");
   683→  const [saving, setSaving] = useState(false);
   684→
   685→  useEffect(() => {
   686→    if (config) setModo(config.modoLiquidacion);
   687→  }, [config, open]);
   688→
   689→  async function handleSave() {
   690→    setSaving(true);
   691→    try {
   692→      await apiPut("/api/configuracion", { modoLiquidacion: modo });
   693→      toast.success("Modo de liquidación actualizado");
   694→      onSaved();
   695→      onOpenChange(false);
   696→    } catch (e) {
   697→      toast.error(e instanceof Error ? e.message : "No se pudo guardar");
   698→    } finally {
   699→      setSaving(false);
   700→    }
   701→  }
   702→
   703→  return (
   704→    <Dialog open={open} onOpenChange={onOpenChange}>
   705→      <DialogContent className="max-w-md">
   706→        <DialogHeader>
   707→          <DialogTitle>Cambiar modo de liquidación</DialogTitle>
   708→          <DialogDescription>
   709→            Elige cómo se reparte la utilidad con el dueño del camión. Puedes configurar los
   710→            detalles en la pestaña Ajustes.
   711→          </DialogDescription>
   712→        </DialogHeader>
   713→        <div className="space-y-3">
   714→          <div className="space-y-2">
   715→            <Label>Modo</Label>
   716→            <Select value={modo} onValueChange={setModo}>
   717→              <SelectTrigger>
   718→                <SelectValue />
   719→              </SelectTrigger>
   720→              <SelectContent>
   721→                {MODOS_LIQUIDACION.map((m) => (
   722→                  <SelectItem key={m.value} value={m.value}>
   723→                    {m.label}
   724→                  </SelectItem>
   725→                ))}
   726→              </SelectContent>
   727→            </Select>
   728→          </div>
   729→          <div className="rounded-lg bg-muted/40 p-3 text-xs text-muted-foreground">
   730→            {modo === "porcentaje" ? (
   731→              <p>La utilidad bruta (ventas − gastos) se reparte por porcentaje entre chofer y dueño.</p>
   732→            ) : (
   733→              <p>El chofer recibe un pago fijo por cada vuelta, dependiendo de si entrega a 1 o múltiples clientes.</p>
   734→            )}
   735→          </div>
   736→        </div>
   737→        <DialogFooter>
   738→          <Button variant="outline" onClick={() => onOpenChange(false)}>
   739→            Cancelar
   740→          </Button>
   741→          <Button onClick={handleSave} disabled={saving}>
   742→            Guardar
   743→          </Button>
   744→        </DialogFooter>
   745→      </DialogContent>
   746→    </Dialog>
   747→  );
   748→}
