   349→  // Helper: aplicar precio según método de pago
   350→  function aplicarPrecioSegunMetodo(metodo: string, cliente: ClienteOption | null) {
   351→    const precioEfectivo = cliente?.precioEfectivoPorLitro || config.precioEfectivoPorLitro || 0;
   352→    const precioTransferencia = cliente?.precioTransferenciaPorLitro || config.precioTransferenciaPorLitro || 0;
   353→    let precio = 0;
   354→    if (metodo === "efectivo") {
   355→      precio = precioEfectivo;
   356→    } else if (metodo === "transferencia") {
   357→      precio = precioTransferencia;
   358→    } else {
   359→      // mixto o crédito (sin pago): usar el precio MÁS ALTO como base
   360→      // — para "sin pago" esto genera la deuda correcta desde el inicio,
   361→      //   y el usuario puede ajustar el precio al momento de cobrar
   362→      precio = Math.max(precioEfectivo, precioTransferencia);
   363→    }
   364→    // Siempre setear el precio (incluso si es 0, para que el usuario vea que no hay configurado)
   365→    setPrecioPorLitro(precio > 0 ? String(precio) : "");
   366→  }

   368→  // Si cambia método de pago (y el usuario no ha tocado el precio manualmente)
   369→  function onChangeMetodoPago(nuevoMetodo: string) {
   370→    const metodoAnterior = metodoPago;
   371→    setMetodoPago(nuevoMetodo);
   372→    // Solo recalcular si el precio actual coincide con el del método anterior
   373→    // (es decir, el usuario no lo modificó manualmente)
   374→    const clienteActual = clienteId ? clientes.find((c) => c.id === clienteId) : null;
   375→    const precioEsperadoAnterior =
   376→      metodoAnterior === "efectivo"
   377→        ? clienteActual?.precioEfectivoPorLitro || config.precioEfectivoPorLitro
   378→        : metodoAnterior === "transferencia"
   379→        ? clienteActual?.precioTransferenciaPorLitro || config.precioTransferenciaPorLitro
   380→        : clienteActual?.precioEfectivoPorLitro || config.precioEfectivoPorLitro;
   381→    const precioActual = Number(precioPorLitro) || 0;
   382→    if (precioActual === precioEsperadoAnterior || precioActual === 0) {
   383→      aplicarPrecioSegunMetodo(nuevoMetodo, clienteActual);
   384→    }
   385→  }
   386→
   387→  // Calcular litros desde lecturas si no se ingresa manualmente
   388→  const litrosCalculado =
   389→    lecturaAnterior !== "" && lecturaActual !== ""
   390→      ? Number(lecturaActual) - Number(lecturaAnterior)
   391→      : null;
   392→
   393→  // === Cálculos con redondeo y descuento ===
   394→  let litrosFinal = litrosManual !== "" ? Number(litrosManual) : litrosCalculado ?? 0;
   395→  const precioFinal = Number(precioPorLitro) || 0;
   396→
   397→  // Modo "por precio": calcular litros desde el precio total
   398→  if (modoIngreso === "por_precio" && precioTotalInput) {
   399→    const precioTotal = Number(precioTotalInput) || 0;
   400→    if (precioFinal > 0) {
   401→      litrosFinal = Math.round(precioTotal / precioFinal);
   402→    }
   403→  }
   404→
   405→  // Si estamos en modo estanques, sumar litros y total de los estanques
   406→  let litrosEstanquesTotal = 0;
   407→  let totalEstanquesRaw = 0;
   408→  if (modoEstanques && estanquesEntrega.length > 0) {
   409→    for (const est of estanquesEntrega) {
   410→      const litrosEst = Number(est.litros) || 0;
   411→      const precioEst = Number(est.precio) || 0;
   412→      litrosEstanquesTotal += litrosEst;
   413→      totalEstanquesRaw += litrosEst * precioEst;
   414→    }
   415→    // En modo estanques, usar los totales de estanques para el resumen
   416→    litrosFinal = litrosEstanquesTotal;
   417→  }
   418→
   419→  // Total calculado con redondeo a $100 superior
   420→  let totalCalculado = modoEstanques ? totalEstanquesRaw : litrosFinal * precioFinal;
   421→  if (totalCalculado > 0) {
   422→    totalCalculado = Math.ceil(totalCalculado / 100) * 100;
   423→  }
   424→
   425→  // Descuento
   426→  let descuentoMontoFinal = 0;
   427→  if (descuentoTipo === "total") {
   428→    descuentoMontoFinal = Number(descuentoMonto) || 0;
   429→  } else if (descuentoTipo === "por_litro") {
   430→    descuentoMontoFinal = (Number(descuentoPorLitroInput) || 0) * litrosFinal;
   431→  }
   432→  const totalConDescuento = Math.max(0, totalCalculado - descuentoMontoFinal);
   433→
   434→  const efectivoFinal = Number(montoEfectivo) || 0;
   435→  const transferenciaFinal = Number(montoTransferencia) || 0;
   436→  const recibidoFinal = efectivoFinal + transferenciaFinal;
   437→  const saldoFinal = totalConDescuento - recibidoFinal;
   438→
   439→  // Auto-actualizar estadoPago según monto recibido (SIEMPRE automático, no manual)
   440→  useEffect(() => {
   441→    if (recibidoFinal === 0) {
   442→      setEstadoPago("pendiente");
   443→    } else if (recibidoFinal >= totalConDescuento && totalConDescuento > 0) {
   444→      setEstadoPago("pagado");
   445→    } else {
   446→      setEstadoPago("parcial");
   447→    }
   448→  }, [recibidoFinal, totalConDescuento]);
