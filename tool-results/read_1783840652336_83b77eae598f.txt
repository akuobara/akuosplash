    94→    94→
    95→    95→export function HistorialVueltasTab({ refreshKey, onRefresh }: { refreshKey: number; onRefresh: () => void }) {
    96→    96→  const [vueltas, setVueltas] = useState<Vuelta[]>([]);
    97→    97→  const [loading, setLoading] = useState(true);
    98→    98→  const [period, setPeriod] = useState<Period>("semana");
    99→    99→  const [vueltasAbiertas, setVueltasAbiertas] = useState<Set<string>>(new Set());
   100→   100→  const [entregasPorVuelta, setEntregasPorVuelta] = useState<Record<string, EntregaVuelta[]>>({});
   101→   101→  const [cargandoEntregas, setCargandoEntregas] = useState<Set<string>>(new Set());
   102→   102→  const [reabrirConfirm, setReabrirConfirm] = useState<Vuelta | null>(null);
   103→   103→  const [reabriendo, setReabriendo] = useState(false);
   104→   104→  // Cache de todas las entregas para no hacer N+1 requests
   105→   105→  const [cacheEntregas, setCacheEntregas] = useState<EntregaVuelta[] | null>(null);
   106→   106→
   107→   107→  const cargar = useCallback(async () => {
   108→   108→    setLoading(true);
   109→   109→    try {
   110→   110→      const data = await apiGet<{ vueltas: Vuelta[] }>(`/api/vueltas?period=${period}`);
   111→   111→      setVueltas(data.vueltas);
   112→   112→      // Invalidar cache de entregas y entregasPorVuelta al refrescar
   113→   113→      setCacheEntregas(null);
   114→   114→      setEntregasPorVuelta({});
   115→   115→    } catch (e) {
   116→   116→      console.error(e);
   117→   117→    } finally {
   118→   118→      setLoading(false);
   119→   119→    }
   120→   120→  }, [period]);
   121→   121→
   122→   122→  useEffect(() => {
   123→   123→    cargar();
   124→   124→  }, [cargar, refreshKey]);
   125→   125→
   126→   126→  // Agrupar por día
   127→   127→  const vueltasPorDia: Record<string, Vuelta[]> = {};
   128→   128→  for (const v of vueltas) {
   129→   129→    const fecha = v.iniciadaEn.split("T")[0];
   130→   130→    if (!vueltasPorDia[fecha]) vueltasPorDia[fecha] = [];
   131→   131→    vueltasPorDia[fecha].push(v);
   132→   132→  }
   133→   133→  const dias = Object.entries(vueltasPorDia).sort(([a], [b]) => b.localeCompare(a));
   134→   134→
   135→   135→  async function toggleVuelta(vueltaId: string) {
   136→   136→    const nuevasAbiertas = new Set(vueltasAbiertas);
   137→   137→    if (nuevasAbiertas.has(vueltaId)) {
   138→   138→      nuevasAbiertas.delete(vueltaId);
   139→   139→      setVueltasAbiertas(nuevasAbiertas);
   140→   140→      return;
   141→   141→    }
   142→   142→    nuevasAbiertas.add(vueltaId);
   143→   143→    setVueltasAbiertas(nuevasAbiertas);
   144→   144→
   145→   145→    if (!entregasPorVuelta[vueltaId]) {
   146→   146→      const nuevasCargando = new Set(cargandoEntregas);
   147→   147→      nuevasCargando.add(vueltaId);
   148→   148→      setCargandoEntregas(nuevasCargando);
   149→   149→      try {
   150→   150→        // Usar cache si ya se cargaron todas las entregas antes
   151→   151→        let todas: EntregaVuelta[] = cacheEntregas ?? [];
   152→   152→        if (!cacheEntregas) {
   153→   153→          const data = await apiGet<{ entregas: EntregaVuelta[] }>(`/api/entregas?period=todo`);
   154→   154→          todas = data.entregas;
   155→   155→          setCacheEntregas(todas);
   156→   156→        }
   157→   157→        const deVuelta = todas.filter((e: any) => e.vueltaId === vueltaId);
   158→   158→        // Usar functional updater para evitar stale closure
   159→   159→        setEntregasPorVuelta((prev) => ({ ...prev, [vueltaId]: deVuelta }));
   160→   160→      } catch (e) {
   161→   161→        console.error(e);
   162→   162→      } finally {
   163→   163→        // Usar functional updater también aquí
   164→   164→        setCargandoEntregas((prev) => {
   165→   165→          const next = new Set(prev);
   166→   166→          next.delete(vueltaId);
   167→   167→          return next;
   168→   168→        });
   169→   169→      }
   170→   170→    }
   171→   171→  }
   172→   172→
   173→   173→  async function reabrirVuelta() {
   174→   174→    if (!reabrirConfirm) return;
   175→   175→    setReabriendo(true);
   176→   176→    try {
   177→   177→      await apiPost(`/api/vueltas/${reabrirConfirm.id}/reabrir`, {});
   178→   178→      toast.success(`Vuelta #${reabrirConfirm.numero} reabierta — ahora es la vuelta activa`);
   179→   179→      setReabrirConfirm(null);
   180→   180→      onRefresh();
   181→   181→      cargar();
   182→   182→    } catch (e) {
   183→   183→      toast.error(e instanceof Error ? e.message : "No se pudo reabrir la vuelta");
   184→   184→    } finally {
   185→   185→      setReabriendo(false);
   186→   186→    }
   187→   187→  }
   188→   188→
   189→   189→  return (
   190→   190→    <div className="space-y-4">
   191→   191→      {/* Encabezado */}
   192→   192→      <div className="flex flex-col gap-3 sm:flex-row sm:items-center sm:justify-between">
   193→   193→        <div>
   194→   194→          <h2 className="text-lg font-semibold sm:text-xl">Historial de vueltas</h2>
   195→   195→          <p className="text-sm text-muted-foreground">
   196→   196→            Detalle de cada vuelta: clientes, litros, cobros y camión usado.
   197→   197→          </p>
   198→   198→        </div>
   199→   199→        <Select value={period} onValueChange={(v) => setPeriod(v as Period)}>
   200→   200→          <SelectTrigger className="w-[150px]">
   201→   201→            <SelectValue />
   202→   202→          </SelectTrigger>
   203→   203→          <SelectContent>
   204→   204→            {PERIODS.map((p) => (
   205→   205→              <SelectItem key={p.value} value={p.value}>{p.label}</SelectItem>
   206→   206→            ))}
   207→   207→          </SelectContent>
   208→   208→        </Select>
   209→   209→      </div>
   210→   210→
   211→   211→      {/* Resumen del período */}
   212→   212→      {vueltas.length > 0 && (
   213→   213→        <div className="grid grid-cols-2 gap-3 sm:grid-cols-4">
   214→   214→          <Card>
   215→   215→            <CardContent className="p-3">
   216→   216→              <p className="text-xs uppercase text-muted-foreground">Vueltas</p>
   217→   217→              <p className="text-xl font-bold">{vueltas.length}</p>
   218→   218→              {(() => {
   219→   219→                const validas = vueltas.filter((v) => esVueltaCompletada(v.litros, v.capacidadCamion, v.cantidadEntregas)).length;
   220→   220→                const invalidas = vueltas.length - validas;
   221→   221→                if (invalidas > 0) {
   222→   222→                  return (
   223→   223→                    <p className="text-[10px] text-amber-600 dark:text-amber-400">
   224→   224→                      {validas} válidas · {invalidas} sin completar
   225→   225→                    </p>
   226→   226→                  );
   227→   227→                }
   228→   228→                return null;
   229→   229→              })()}
   230→   230→            </CardContent>
   231→   231→          </Card>
   232→   232→          <Card>
   233→   233→            <CardContent className="p-3">
   234→   234→              <p className="text-xs uppercase text-muted-foreground">Litros</p>
   235→   235→              <p className="text-xl font-bold tabular-nums">{formatNumber(vueltas.reduce((s, v) => s + v.litros, 0), 0)} L</p>
   236→   236→            </CardContent>
   237→   237→          </Card>
   238→   238→          <Card>
   239→   239→            <CardContent className="p-3">
   240→   240→              <p className="text-xs uppercase text-muted-foreground">Ventas</p>
   241→   241→              <p className="text-xl font-bold tabular-nums">{formatCurrency(vueltas.reduce((s, v) => s + v.ventas, 0))}</p>
   242→   242→            </CardContent>
   243→   243→          </Card>
   244→   244→          <Card>
   245→   245→            <CardContent className="p-3">
   246→   246→              <p className="text-xs uppercase text-muted-foreground">Efectivo</p>
   247→   247→              <p className="text-xl font-bold tabular-nums text-emerald-600 dark:text-emerald-400">
   248→   248→                {formatCurrency(vueltas.reduce((s, v) => s + v.efectivoRecibido, 0))}
   249→   249→              </p>
   250→   250→            </CardContent>
   251→   251→          </Card>
   252→   252→        </div>
   253→   253→      )}
   254→   254→
   255→   255→      {/* Lista por día */}
   256→   256→      {loading ? (
   257→   257→        <div className="space-y-2">
   258→   258→          {Array.from({ length: 3 }).map((_, i) => <Skeleton key={i} className="h-24 rounded-lg" />)}
   259→   259→        </div>
   260→   260→      ) : vueltas.length === 0 ? (
   261→   261→        <Card>
   262→   262→          <CardContent className="flex flex-col items-center justify-center gap-2 p-10 text-center">
   263→   263→            <Repeat className="h-10 w-10 text-muted-foreground" />
   264→   264→            <p className="text-sm text-muted-foreground">No hay vueltas en este período.</p>
   265→   265→          </CardContent>
   266→   266→        </Card>
   267→   267→      ) : (
   268→   268→        <div className="space-y-4">
   269→   269→          {dias.map(([fechaKey, vueltasDia]) => {
   270→   270→            const f = new Date(fechaKey + "T12:00:00");
   271→   271→            const hoy = new Date();
   272→   272→            hoy.setHours(0, 0, 0, 0);
   273→   273→            const ayer = new Date(hoy);
   274→   274→            ayer.setDate(ayer.getDate() - 1);
   275→   275→            const esHoy = f.toDateString() === hoy.toDateString();
   276→   276→            const esAyer = f.toDateString() === ayer.toDateString();
   277→   277→            const etiqueta = esHoy ? "Hoy" : esAyer ? "Ayer" : formatDate(f);
   278→   278→
   279→   279→            const litrosDia = vueltasDia.reduce((s, v) => s + v.litros, 0);
   280→   280→            const ventasDia = vueltasDia.reduce((s, v) => s + v.ventas, 0);
   281→   281→            const efectivoDia = vueltasDia.reduce((s, v) => s + v.efectivoRecibido, 0);
   282→   282→
   283→   283→            return (
   284→   284→              <div key={fechaKey} className="space-y-2">
   285→   285→                {/* Encabezado del día */}
   286→   286→                <div className="flex items-center justify-between rounded-lg bg-muted/40 px-3 py-2">
   287→   287→                  <div>
   288→   288→                    <span className="text-sm font-semibold">{etiqueta}</span>
   289→   289→                    <span className="ml-2 text-xs text-muted-foreground">{vueltasDia.length} vuelta(s)</span>
   290→   290→                  </div>
   291→   291→                  <div className="flex items-center gap-3 text-xs">
   292→   292→                    <span className="tabular-nums">{formatNumber(litrosDia, 0)} L</span>
   293→   293→                    <span className="tabular-nums font-semibold">{formatCurrency(ventasDia)}</span>
   294→   294→                    <span className="tabular-nums text-emerald-600 dark:text-emerald-400">{formatCurrency(efectivoDia)}</span>
   295→   295→                  </div>
   296→   296→                </div>
   297→   297→
   298→   298→                {/* Vueltas del día */}
   299→   299→                {vueltasDia.map((v) => {
   300→   300→                  const abierta = vueltasAbiertas.has(v.id);
   301→   301→                  const entregas = entregasPorVuelta[v.id] || [];
   302→   302→                  const cargando = cargandoEntregas.has(v.id);
   303→   303→                  const completada = esVueltaCompletada(v.litros, v.capacidadCamion, v.cantidadEntregas);
   304→   304→                  const porcentaje = v.capacidadCamion > 0 ? (v.litros / v.capacidadCamion) * 100 : 0;
   305→   305→                  const umbralLitros = v.capacidadCamion > 0 ? Math.ceil(v.capacidadCamion * UMBRAL_COMPLETACION_VUELTA) : 0;
   306→   306→                  const puedeReabrir = !v.activa && !completada;
   307→   307→
   308→   308→                  return (
   309→   309→                    <Card key={v.id} className={`overflow-hidden ${!completada && !v.activa ? "border-amber-300 dark:border-amber-800" : ""}`}>
   310→   310→                      <button
   311→   311→                        type="button"
   312→   312→                        onClick={() => toggleVuelta(v.id)}
   313→   313→                        className="flex w-full items-center gap-3 p-3 text-left hover:bg-muted/30"
   314→   314→                      >
   315→   315→                        <div className={`flex h-10 w-10 shrink-0 items-center justify-center rounded-lg font-bold ${
   316→   316→                          v.activa
   317→   317→                            ? "bg-primary text-primary-foreground"
   318→   318→                            : completada
   319→   319→                              ? "bg-emerald-100 text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-300"
   320→   320→                              : "bg-amber-100 text-amber-700 dark:bg-amber-900/30 dark:text-amber-300"
   321→   321→                        }`}>
   322→   322→                          #{v.numero}
   323→   323→                        </div>
   324→   324→                        <div className="min-w-0 flex-1">
   325→   325→                          <div className="flex flex-wrap items-center gap-1.5">
   326→   326→                            <span className="font-medium text-sm">Vuelta #{v.numero}</span>
   327→   327→                            {v.activa ? (
   328→   328→                              <span className="inline-flex items-center gap-0.5 rounded-full bg-emerald-100 px-1.5 py-0.5 text-[10px] font-medium text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-300">
   329→   329→                                <span className="h-1.5 w-1.5 rounded-full bg-emerald-500" /> Activa
   330→   330→                              </span>
   331→   331→                            ) : completada ? (
   332→   332→                              <span className="inline-flex items-center gap-0.5 rounded-full bg-emerald-100 px-1.5 py-0.5 text-[10px] font-medium text-emerald-700 dark:bg-emerald-900/30 dark:text-emerald-300">
   333→   333→                                <CheckCircle2 className="h-2.5 w-2.5" /> Completada
   334→   334→                              </span>
   335→   335→                            ) : (
   336→   336→                              <span className="inline-flex items-center gap-0.5 rounded-full bg-amber-100 px-1.5 py-0.5 text-[10px] font-medium text-amber-700 dark:bg-amber-900/30 dark:text-amber-300">
   337→   337→                                <AlertTriangle className="h-2.5 w-2.5" /> Sin completar
   338→   338→                              </span>
   339→   339→                            )}
   340→   340→                            {v.esMultiple && (
   341→   341→                              <Badge variant="outline" className="text-[10px]">Múltiples</Badge>
   342→   342→                            )}
   343→   343→                            {v.camion && (
   344→   344→                              <Badge variant="outline" className="text-[10px] gap-0.5">
   345→   345→                                <Truck className="h-2.5 w-2.5" /> {v.camion.nombre}
   346→   346→                              </Badge>
   347→   347→                            )}
   348→   348→                            {v.chofer && (
   349→   349→                              <Badge variant="outline" className="text-[10px]">{v.chofer.nombre}</Badge>
   350→   350→                            )}
   351→   351→                          </div>
   352→   352→                          <div className="mt-0.5 flex flex-wrap items-center gap-2 text-xs text-muted-foreground">
   353→   353→                            <span className="flex items-center gap-0.5">
   354→   354→                              <Clock className="h-3 w-3" /> {formatTime(v.iniciadaEn)}
   355→   355→                              {v.cerradaEn ? ` → ${formatTime(v.cerradaEn)}` : " · en curso"}
   356→   356→                            </span>
   357→   357→                            <span>·</span>
   358→   358→                            <span><Droplets className="mr-0.5 inline h-3 w-3" />{formatNumber(v.litros, 0)} L</span>
   359→   359→                            {v.cantidadEntregas > 0 && <><span>·</span><span>{v.cantidadEntregas} entrega(s)</span></>}
   360→   360→                            {v.capacidadCamion > 0 && (
   361→   361→                              <span className={completada ? "text-emerald-600 dark:text-emerald-400" : "text-amber-600 dark:text-amber-400"}>
   362→   362→                                · {porcentaje.toFixed(0)}% de {formatNumber(v.capacidadCamion, 0)} L
   363→   363→                              </span>
   364→   364→                            )}
   365→   365→                          </div>
   366→   366→                        </div>
   367→   367→                        <div className="shrink-0 text-right">
   368→   368→                          {v.ventas > 0 && <p className="font-semibold tabular-nums text-sm">{formatCurrency(v.ventas)}</p>}
   369→   369→                          {v.efectivoRecibido > 0 && <p className="text-xs text-emerald-600 dark:text-emerald-400 tabular-nums">Efect: {formatCurrency(v.efectivoRecibido)}</p>}
   370→   370→                        </div>
   371→   371→                        {abierta ? <ChevronDown className="h-4 w-4 shrink-0 text-muted-foreground" /> : <ChevronRight className="h-4 w-4 shrink-0 text-muted-foreground" />}
   372→   372→                      </button>
   373→   373→
   374→   374→                      {/* Banner de vuelta no completada con botón Retomar */}
   375→   375→                      {puedeReabrir && (
   376→   376→                        <div className="border-t border-amber-200 bg-amber-50 px-3 py-2 dark:border-amber-800 dark:bg-amber-900/20">
   377→   377→                          <div className="flex flex-col gap-2 sm:flex-row sm:items-center sm:justify-between">
   378→   378→                            <p className="text-xs text-amber-900 dark:text-amber-200">
   379→   379→                              <AlertTriangle className="mr-1 inline h-3 w-3" />
   380→   380→                              Vuelta cerrada sin completar ({porcentaje.toFixed(0)}% de {formatNumber(v.capacidadCamion, 0)} L).
   381→   381→                              {" "}Faltaron {formatNumber(Math.max(0, umbralLitros - v.litros), 0)} L para ser válida.
   382→   382→                              {" "}No cuenta para la liquidación.
   383→   383→                            </p>
   384→   384→                            <Button
   385→   385→                              size="sm"
