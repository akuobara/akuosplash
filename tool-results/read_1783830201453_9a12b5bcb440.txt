   613→   613→function ConfigModal({

   616→   616→  config,
   617→   617→  onSaved,
   618→   618→}: {
   619→   619→  open: boolean;

   621→   621→  config: Config | null;
   622→   622→  onSaved: () => void;
   623→   623→}) {
   624→   624→  const [modo, setModo] = useState("porcentaje");
   625→   625→  const [saving, setSaving] = useState(false);
   626→   626→
   627→   627→  useEffect(() => {
   628→   628→    if (config) setModo(config.modoLiquidacion);
   629→   629→  }, [config, open]);
   630→   630→
   631→   631→  async function handleSave() {
   632→   632→    setSaving(true);
   633→   633→    try {
   634→   634→      await apiPut("/api/configuracion", { modoLiquidacion: modo });
   635→   635→      toast.success("Modo de liquidación actualizado");

   641→   641→      setSaving(false);
   642→   642→    }

   645→   645→  return (
   646→   646→    <Dialog open={open} onOpenChange={onOpenChange}>

   652→   652→            detalles en la pestaña Ajustes.
   653→   653→          </DialogDescription>
   654→   654→        </DialogHeader>

   659→   659→              <SelectTrigger>
   660→   660→                <SelectValue />

   672→   672→            {modo === "porcentaje" ? (
   673→   673→              <p>La utilidad bruta (ventas − gastos) se reparte por porcentaje entre chofer y dueño.</p>
   674→   674→            ) : (

   681→   681→            Cancelar

   684→   684→            Guardar
   685→   685→          </Button>
   686→   686→        </DialogFooter>
   687→   687→      </DialogContent>
   688→   688→    </Dialog>
