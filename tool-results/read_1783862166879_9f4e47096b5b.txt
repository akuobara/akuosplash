   334→  // Helper: aplicar precio según método de pago
   335→  function aplicarPrecioSegunMetodo(metodo: string, cliente: ClienteOption | null) {
   336→    const precioEfectivo = cliente?.precioEfectivoPorLitro || config.precioEfectivoPorLitro || 0;
   337→    const precioTransferencia = cliente?.precioTransferenciaPorLitro || config.precioTransferenciaPorLitro || 0;
   338→    let precio = 0;
   339→    if (metodo === "efectivo") {
   340→      precio = precioEfectivo;
   341→    } else if (metodo === "transferencia") {
   342→      precio = precioTransferencia;
   343→    } else {
   344→      // mixto o crédito (sin pago): usar el precio MÁS ALTO como base
   345→      // — para "sin pago" esto genera la deuda correcta desde el inicio,
   346→      //   y el usuario puede ajustar el precio al momento de cobrar
   347→      precio = Math.max(precioEfectivo, precioTransferencia);
   348→    }
   349→    // Siempre setear el precio (incluso si es 0, para que el usuario vea que no hay configurado)
   350→    setPrecioPorLitro(precio > 0 ? String(precio) : "");
   351→  }

   363→        : metodoAnterior === "transferencia"

   410→  // Guarda la entrega con un vueltaId opcional (cuando se inicia/reabre desde el diálogo)
   411→  async function guardarEntrega(vueltaIdOverride?: string) {
   412→    // Capturar el estado actual en variables locales para evitar problemas de stale closure
   413→    const modoEst = modoEstanques;
   414→    const estanques = estanquesEntrega;
   415→    const clienteNombreLocal = clienteNombre.trim();
   416→    const litrosFinalLocal = litrosFinal;
   417→    const precioFinalLocal = precioFinal;
   418→    const totalCalculadoLocal = totalCalculado;
   419→    const efectivoFinalLocal = efectivoFinal;
   420→    const transferenciaFinalLocal = transferenciaFinal;
   421→
   422→    // Determinar si hay datos de estanques con litros > 0
   423→    const estanquesConLitros = estanques.filter((e) => Number(e.litros) > 0);
   424→
   425→    // Si está en modo estanques (o hay estanques con litros), usar el path de estanques
   426→    if ((modoEst || estanquesConLitros.length > 0) && estanques.length > 0) {
   427→      const totalLitrosEst = estanquesConLitros.reduce((s, e) => s + (Number(e.litros) || 0), 0);
   428→      if (totalLitrosEst <= 0) {
   429→        toast.error("Ingresa al menos litros en un estanque");
   430→        return;
   431→      }
   432→
   433→      // Normalizar montoRecibido por estanque: si es efectivo/transferencia y está vacío,
   434→      // usar el total calculado (litros × precio). Si es credito, queda 0 (deuda).
   435→      const estanquesNormalizados = estanquesConLitros.map((e) => {
   436→        const litrosNum = Number(e.litros) || 0;
   437→        const precioNum = Number(e.precio) || 0;
   438→        const totalEstCalc = litrosNum * precioNum;
   439→        let montoRecibidoNum = Number(e.montoRecibido) || 0;
   440→        if (e.metodoPago !== "credito" && montoRecibidoNum === 0 && totalEstCalc > 0) {
   441→          // Auto-llenar montoRecibido con el total si el usuario no lo editó manualmente
   442→          montoRecibidoNum = Math.round(totalEstCalc);
   443→        }
   444→        return {
   445→          ...e,
   446→          litros: litrosNum,
   447→          precio: precioNum,
   448→          montoRecibido: montoRecibidoNum,
   449→          total: totalEstCalc,
   450→        };
   451→      });
   452→
   453→      const totalVentaEst = estanquesNormalizados.reduce((s, e) => s + e.total, 0);
   454→      const totalEfectivoEst = estanquesNormalizados.filter((e) => e.metodoPago === "efectivo").reduce((s, e) => s + e.montoRecibido, 0);
   455→      const totalTransferenciaEst = estanquesNormalizados.filter((e) => e.metodoPago === "transferencia").reduce((s, e) => s + e.montoRecibido, 0);
   456→      const totalRecibidoEst = totalEfectivoEst + totalTransferenciaEst;
   457→
   458→      let estadoPagoEst = "pendiente";
   459→      if (totalRecibidoEst >= totalVentaEst && totalVentaEst > 0) estadoPagoEst = "pagado";
   460→      else if (totalRecibidoEst > 0) estadoPagoEst = "parcial";
   461→
   462→      setSaving(true);
   463→      try {
   464→        const payload: Record<string, unknown> = {
   465→          clienteId: clienteId && clienteId !== "__manual" ? clienteId : null,
   466→          clienteNombre: clienteNombreLocal,

   597→  async function iniciarNuevaYGuardar() {

   601→    const camionesActivos = (camiones ?? []).filter((c) => c.activo);

   607→    setIniciandoVuelta(true);
   608→    try {
   609→      const res = await apiPost<{ vuelta: { id: string; numero: number } }>("/api/vueltas", { camionId });
   610→      toast.success(`Vuelta #${res.vuelta.numero} iniciada`);
   611→      // Guardar la entrega PRIMERO (sin notificar al padre aún para evitar re-renders que puedan afectar el estado)
   612→      await guardarEntrega(res.vuelta.id);

   632→      // Después de guardar, onSaved() ya se encargó de refrescar el padre
   633→    } catch (e) {
   634→      toast.error(e instanceof Error ? e.message : "No se pudo reabrir la vuelta");
   635→    } finally {
   636→      setIniciandoVuelta(false);

   640→  async function handleSubmit(e: React.FormEvent) {
   641→    e.preventDefault();

   659→      return;

   661→
   662→    // Calcular litros totales según modo
   663→    let litrosAEntregar = litrosFinal;
   664→    if (modoEstanques && estanquesEntrega.length > 0) {
   665→      litrosAEntregar = estanquesEntrega.reduce((s, e) => s + (Number(e.litros) || 0), 0);

   668→    // Avisar (no bloquear) si excede la capacidad del camión

   677→          );

   681→            `Cierra la vuelta y empieza una nueva para recargar.`,
   682→            { duration: 8000 }
   683→          );
   684→        }
   685→      } else if (disponible - litrosAEntregar <= tolerancia && disponible - litrosAEntregar > 0) {

   687→          `Después de esta entrega quedarán ${formatNumber(disponible - litrosAEntregar, 0)} L en el camión. ` +
   688→          `Considera cerrar la vuelta pronto.`,
   689→          { duration: 6000 }
   690→        );
   691→      }
   692→    }

   694→    // Guardar con la vuelta activa actual
   695→    await guardarEntrega();
