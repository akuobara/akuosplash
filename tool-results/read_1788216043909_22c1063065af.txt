   241→  async function soltarCliente(clienteId: string, clienteNombre: string) {
   242→    const reemplazo = reemplazoChoferId && reemplazoChoferId !== "__ninguno" ? reemplazoChoferId : null;

   244→      ? `¿Soltar a ${clienteNombre} y asignar a ${choferes.find(c => c.id === reemplazo)?.nombre ?? "otro chofer"} como reemplazo?`

   298→  // Capturar ubicación GPS actual del dispositivo
   299→  function obtenerMiUbicacion() {
   300→    if (!navigator.geolocation) {
   301→      toast.error("Tu navegador no soporta geolocalización");
   302→      return;
   303→    }
   304→    setObteniendoGPS(true);
   305→    navigator.geolocation.getCurrentPosition(
   306→      (pos) => {
   307→        setLat(pos.coords.latitude.toFixed(6));
   308→        setLng(pos.coords.longitude.toFixed(6));
   309→        setObteniendoGPS(false);
   310→        toast.success("Ubicación capturada");
   311→      },
   312→      (err) => {
   313→        setObteniendoGPS(false);
   314→        toast.error(err.message || "No se pudo obtener tu ubicación");
   315→      },
   316→      { enableHighAccuracy: true, timeout: 10000, maximumAge: 0 }
   317→    );

   354→        await apiPost("/api/clientes", payload);

   356→      }
   357→      setFormOpen(false);
   358→      onRefresh();
   359→    } catch (e) {
   360→      toast.error(e instanceof Error ? e.message : "No se pudo guardar");
