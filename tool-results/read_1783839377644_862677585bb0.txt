   334→  // Helper: aplicar precio según método de pago
   335→  function aplicarPrecioSegunMetodo(metodo: string, cliente: ClienteOption | null) {
   336→    const precioEfectivo = cliente?.precioEfectivoPorLitro || config.precioEfectivoPorLitro || 0;
   337→    const precioTransferencia = cliente?.precioTransferenciaPorLitro || config.precioTransferenciaPorLitro || 0;
   338→    let precio = 0;
   339→    if (metodo === "efectivo") {
   340→      precio = precioEfectivo;
   341→    } else if (metodo === "transferencia") {
   342→      precio = precioTransferencia;
   343→    } else {
   344→      // mixto o crédito (sin pago): usar el precio MÁS ALTO como base
   345→      // — para "sin pago" esto genera la deuda correcta desde el inicio,
   346→      //   y el usuario puede ajustar el precio al momento de cobrar
   347→      precio = Math.max(precioEfectivo, precioTransferencia);
   348→    }
   349→    // Siempre setear el precio (incluso si es 0, para que el usuario vea que no hay configurado)
   350→    setPrecioPorLitro(precio > 0 ? String(precio) : "");
   351→  }

   360→    const precioEsperadoAnterior =
   361→      metodoAnterior === "efectivo"
   362→        ? clienteActual?.precioEfectivoPorLitro || config.precioEfectivoPorLitro
   363→        : metodoAnterior === "transferencia"
   364→        ? clienteActual?.precioTransferenciaPorLitro || config.precioTransferenciaPorLitro
   365→        : clienteActual?.precioEfectivoPorLitro || config.precioEfectivoPorLitro;
   366→    const precioActual = Number(precioPorLitro) || 0;
   367→    if (precioActual === precioEsperadoAnterior || precioActual === 0) {
   368→      aplicarPrecioSegunMetodo(nuevoMetodo, clienteActual);
   369→    }
   370→  }

   374→    lecturaAnterior !== "" && lecturaActual !== ""

   376→      : null;
   377→
   378→  const litrosFinal = litrosManual !== "" ? Number(litrosManual) : litrosCalculado ?? 0;

   380→  const totalCalculado = litrosFinal * precioFinal;
   381→  const efectivoFinal = Number(montoEfectivo) || 0;
   382→  const transferenciaFinal = Number(montoTransferencia) || 0;
   383→  const recibidoFinal = efectivoFinal + transferenciaFinal;

   385→

   389→      setEstadoPago("pendiente");
   390→    } else if (recibidoFinal >= totalCalculado && totalCalculado > 0) {
   391→      setEstadoPago("pagado");
   392→    } else {
   393→      setEstadoPago("parcial");
   394→    }
   395→  }, [recibidoFinal, totalCalculado]);

   410→  // Guarda la entrega con un vueltaId opcional (cuando se inicia/reabre desde el diálogo)
   411→  async function guardarEntrega(vueltaIdOverride?: string) {
   412→    // Capturar el estado actual en variables locales para evitar problemas de stale closure
   413→    const modoEst = modoEstanques;
   414→    const estanques = estanquesEntrega;
   415→    const clienteNombreLocal = clienteNombre.trim();
   416→    const litrosFinalLocal = litrosFinal;
   417→    const precioFinalLocal = precioFinal;
   418→    const totalCalculadoLocal = totalCalculado;
   419→    const efectivoFinalLocal = efectivoFinal;
   420→    const transferenciaFinalLocal = transferenciaFinal;
   421→
   422→    // Determinar si hay datos de estanques con litros > 0
   423→    const estanquesConLitros = estanques.filter((e) => Number(e.litros) > 0);
   424→
   425→    // Si está en modo estanques (o hay estanques con litros), usar el path de estanques
   426→    if ((modoEst || estanquesConLitros.length > 0) && estanques.length > 0) {
   427→      const totalLitrosEst = estanquesConLitros.reduce((s, e) => s + (Number(e.litros) || 0), 0);
   428→      if (totalLitrosEst <= 0) {
   429→        toast.error("Ingresa al menos litros en un estanque");
   430→        return;
   431→      }
   432→      const totalVentaEst = estanquesConLitros.reduce((s, e) => s + (Number(e.litros) || 0) * (Number(e.precio) || 0), 0);
   433→      const totalEfectivoEst = estanquesConLitros.filter((e) => e.metodoPago === "efectivo").reduce((s, e) => s + (Number(e.montoRecibido) || 0), 0);
   434→      const totalTransferenciaEst = estanquesConLitros.filter((e) => e.metodoPago === "transferencia").reduce((s, e) => s + (Number(e.montoRecibido) || 0), 0);
   435→      const totalRecibidoEst = totalEfectivoEst + totalTransferenciaEst;
   436→
   437→      let estadoPagoEst = "pendiente";
   438→      if (totalRecibidoEst >= totalVentaEst && totalVentaEst > 0) estadoPagoEst = "pagado";
   439→      else if (totalRecibidoEst > 0) estadoPagoEst = "parcial";
   440→
   441→      setSaving(true);
   442→      try {
   443→        const payload: Record<string, unknown> = {
   444→          clienteId: clienteId && clienteId !== "__manual" ? clienteId : null,
   445→          clienteNombre: clienteNombreLocal,
   446→          clienteDireccion: clienteDireccion.trim() || null,
   447→          clienteTelefono: clienteTelefono.trim() || null,
   448→          clienteZona: clienteZona.trim() || null,
   449→          litros: totalLitrosEst,
   450→          metodoPago: "mixto",
   451→          precioPorLitro: 0,
   452→          total: totalVentaEst,
   453→          montoEfectivo: totalEfectivoEst,
   454→          montoTransferencia: totalTransferenciaEst,
   455→          estadoPago: estadoPagoEst,
   456→          fotoMedidorUrl: fotoUrl,
   457→          nota: nota.trim() || null,
   458→          detalleEstanquesJson: JSON.stringify(estanquesConLitros.map((e) => ({
   459→            nombre: e.nombre,
   460→            litros: Number(e.litros) || 0,
   461→            precioPorLitro: Number(e.precio) || 0,
   462→            total: (Number(e.litros) || 0) * (Number(e.precio) || 0),
   463→            metodoPago: e.metodoPago,

   619→  async function handleSubmit(e: React.FormEvent) {
