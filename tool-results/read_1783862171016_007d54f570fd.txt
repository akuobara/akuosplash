   335→   335→  function aplicarPrecioSegunMetodo(metodo: string, cliente: ClienteOption | null) {
   336→   336→    const precioEfectivo = cliente?.precioEfectivoPorLitro || config.precioEfectivoPorLitro || 0;
   337→   337→    const precioTransferencia = cliente?.precioTransferenciaPorLitro || config.precioTransferenciaPorLitro || 0;
   338→   338→    let precio = 0;
   339→   339→    if (metodo === "efectivo") {
   340→   340→      precio = precioEfectivo;
   341→   341→    } else if (metodo === "transferencia") {
   342→   342→      precio = precioTransferencia;
   343→   343→    } else {
   344→   344→      // mixto o crédito (sin pago): usar el precio MÁS ALTO como base
   345→   345→      // — para "sin pago" esto genera la deuda correcta desde el inicio,
   346→   346→      //   y el usuario puede ajustar el precio al momento de cobrar
   347→   347→      precio = Math.max(precioEfectivo, precioTransferencia);
   348→   348→    }
   349→   349→    // Siempre setear el precio (incluso si es 0, para que el usuario vea que no hay configurado)
   350→   350→    setPrecioPorLitro(precio > 0 ? String(precio) : "");
   351→   351→  }

   354→   354→  function onChangeMetodoPago(nuevoMetodo: string) {
   355→   355→    const metodoAnterior = metodoPago;
   356→   356→    setMetodoPago(nuevoMetodo);
   357→   357→    // Solo recalcular si el precio actual coincide con el del método anterior
   358→   358→    // (es decir, el usuario no lo modificó manualmente)
   359→   359→    const clienteActual = clienteId ? clientes.find((c) => c.id === clienteId) : null;
   360→   360→    const precioEsperadoAnterior =
   361→   361→      metodoAnterior === "efectivo"
   362→   362→        ? clienteActual?.precioEfectivoPorLitro || config.precioEfectivoPorLitro
   363→   363→        : metodoAnterior === "transferencia"
   364→   364→        ? clienteActual?.precioTransferenciaPorLitro || config.precioTransferenciaPorLitro
   365→   365→        : clienteActual?.precioEfectivoPorLitro || config.precioEfectivoPorLitro;
   366→   366→    const precioActual = Number(precioPorLitro) || 0;
   367→   367→    if (precioActual === precioEsperadoAnterior || precioActual === 0) {
   368→   368→      aplicarPrecioSegunMetodo(nuevoMetodo, clienteActual);
   369→   369→    }
   370→   370→  }
   371→   371→
   372→   372→  // Calcular litros desde lecturas si no se ingresa manualmente
   373→   373→  const litrosCalculado =
   374→   374→    lecturaAnterior !== "" && lecturaActual !== ""
   375→   375→      ? Number(lecturaActual) - Number(lecturaAnterior)
   376→   376→      : null;
   377→   377→
   378→   378→  const litrosFinal = litrosManual !== "" ? Number(litrosManual) : litrosCalculado ?? 0;
   379→   379→  const precioFinal = Number(precioPorLitro) || 0;
   380→   380→  const totalCalculado = litrosFinal * precioFinal;
   381→   381→  const efectivoFinal = Number(montoEfectivo) || 0;
   382→   382→  const transferenciaFinal = Number(montoTransferencia) || 0;
   383→   383→  const recibidoFinal = efectivoFinal + transferenciaFinal;
   384→   384→  const saldoFinal = totalCalculado - recibidoFinal;
   385→   385→
   386→   386→  // Auto-actualizar estadoPago según monto recibido (SIEMPRE automático, no manual)
   387→   387→  useEffect(() => {
   388→   388→    if (recibidoFinal === 0) {
   389→   389→      setEstadoPago("pendiente");
   390→   390→    } else if (recibidoFinal >= totalCalculado && totalCalculado > 0) {
   391→   391→      setEstadoPago("pagado");
   392→   392→    } else {
   393→   393→      setEstadoPago("parcial");
   394→   394→    }
   395→   395→  }, [recibidoFinal, totalCalculado]);

   397→   397→  async function handleFoto(file: File) {
